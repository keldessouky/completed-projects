# The ROM, played on the real thing: DeSmuME running dist/crawler-ds.nds with
# nothing going in but button presses and stylus taps, and the game's own
# telemetry read back out of emulated main RAM.

echo boot
frames 150
expect scene = 0
expect carl_hp > 0
expect floor = 1
shot 01-title

# The game opens above ground now: chapter one is a couple of dozen lines with
# questions in it, so getting to the first corridor takes a lot more taps than
# a briefing did.
echo chapter one
until a scene = 11 12
frames 20
expect scene = 11
shot 02-chapter
until a scene = 2 260
frames 30
expect explored > 8
shot 03-corridor

# The stylus test goes first, at the entrance, where it is deterministic: the
# floor is generated per season now, but the party always arrives facing a
# tile it can walk onto, so a tap on the touch d-pad has to move it.
echo the stylus alone can play it
touch 48 151 12
touch 48 151 12
frames 20
expect steps > 0

# Everything after this has to be map-agnostic for the same reason. autoplay
# walks, turns away from walls and confirms whatever the game puts on screen,
# so it does not care what the season dealt.
#  A run now owes the party a handful of loot boxes for walking in at all, and
#  each one is a scene to tap through, so the budget covers the drip.
echo walking the floor
autoplay 30
expect steps > 3

# autoplay can leave the party mid-fight, and a fight is not over in twenty
# taps, so the budget here has to be big enough to finish one.
echo the party screen
until a scene = 2 80
touch 214 128 12
frames 25
expect scene = 4
shot 03-party
press b 10
frames 20
expect scene = 2

echo explore, fight, loot
autoplay 520
expect steps > 20
expect explored > 40
expect battles_won >= 1
expect carl_hp > 0
frames 20
shot 04-play
print

echo keep going
autoplay 520
expect steps > 20
expect explored > 40
expect boxes >= 1
expect carl_hp > 0
print
shot 05-later

echo the ROM is still running and has not wedged
expect frame > 3000
expect scene != 9

cmake_minimum_required(VERSION 3.20)
project(qahira CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

file(GLOB_RECURSE CORE_SRC CONFIGURE_DEPENDS src/*.cpp)
add_library(qahira_obj OBJECT ${CORE_SRC} third_party/stb_impl.cpp)
target_include_directories(qahira_obj PUBLIC src third_party/libretro third_party/stb)
target_compile_options(qahira_obj PRIVATE -Wall -Wno-unused-function -Wno-missing-braces)
if(ANDROID)
  target_compile_options(qahira_obj PRIVATE -O3 -ffast-math -fno-exceptions)
endif()

add_library(qahira_libretro SHARED $<TARGET_OBJECTS:qahira_obj>)
set_target_properties(qahira_libretro PROPERTIES PREFIX "")
if(ANDROID)
  set_target_properties(qahira_libretro PROPERTIES OUTPUT_NAME "qahira_libretro_android")
  target_link_libraries(qahira_libretro PRIVATE GLESv3 EGL log android m)
  target_link_options(qahira_libretro PRIVATE -Wl,--no-undefined -static-libstdc++)
elseif(APPLE)
  target_link_libraries(qahira_libretro PRIVATE "-framework OpenGL")
endif()

option(QAHIRA_HOST "Build the SDL dev host (qhost)" ON)
if(NOT ANDROID AND QAHIRA_HOST)
  find_package(SDL2 CONFIG REQUIRED)
  add_executable(qhost host/qhost.cpp $<TARGET_OBJECTS:qahira_obj>)
  target_include_directories(qhost PRIVATE src third_party/libretro third_party/stb)
  target_link_libraries(qhost PRIVATE SDL2::SDL2)
  if(APPLE)
    target_link_libraries(qhost PRIVATE "-framework OpenGL")
  endif()
endif()

if(NOT ANDROID)
  file(GLOB TEST_SRC CONFIGURE_DEPENDS tests/*.cpp)
  add_executable(qtests ${TEST_SRC} $<TARGET_OBJECTS:qahira_obj>)
  target_include_directories(qtests PRIVATE . src third_party/libretro third_party/stb)
  target_compile_definitions(qtests PRIVATE QAHIRA_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  if(APPLE)
    target_link_libraries(qtests PRIVATE "-framework OpenGL")
  endif()
endif()
